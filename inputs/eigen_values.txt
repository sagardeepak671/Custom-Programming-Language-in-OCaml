// Eigenvalues of a 2x2 matrix using its trace and determinant.
floatmatrix 2,2 a := [[2.,1.],[1.,2.]];
float trace := a[0,0] + a[1,1];
float det := determinant(a);
float discriminant := trace * trace - 4. * det;
if (discriminant >= 0.) then {
    float root := 0.;
    if (discriminant > 0.) then {
        float guess := discriminant;
        for (i := 0, i < 20, 1) {
            guess := (guess + discriminant / guess) / 2.;
        };
        root := guess;
    };
    Print((trace + root) / 2.);
    Print((trace - root) / 2.);
} else {
    Print(false);
};
