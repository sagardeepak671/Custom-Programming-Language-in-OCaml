// Recursion: the naive recursive Fibonacci function checked against an iterative one.
function int fib(int n) {
    if (n < 2) then {
        return n;
    };
    return fib(n - 1) + fib(n - 2);
};

function int fib_loop(int n) {
    int a := 0;
    int b := 1;
    for (i := 0, i < n, 1) {
        int next := a + b;
        a := b;
        b := next;
    };
    return a;
};

for (i := 0, i <= 10, 1) {
    Print(fib(i));
};
Print(fib(20) == fib_loop(20));
Print(fib_loop(90));
